rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // progresso de cada jogador: só o dono lê e escreve
    match /usuarios/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    // salas de partidas online: quem tem conta pode criar e entrar;
    // depois que a sala fecha, só os dois jogadores podem mexer nela
    match /salas/{codigo} {
      allow read, create: if request.auth != null;
      allow update: if request.auth != null && (
        resource.data.status == 'esperando' ||
        request.auth.uid == resource.data.brancasUid ||
        request.auth.uid == resource.data.pretasUid);
    }
  }
}
