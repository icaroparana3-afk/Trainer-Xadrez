rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // progresso de cada jogador: só o dono lê e escreve
    match /usuarios/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    // ranking: qualquer um vê (nome, foto, nível, XP e pontos);
    // cada jogador só mexe na própria linha
    match /ranking/{uid} {
      allow read: if true;
      allow delete: if request.auth != null && request.auth.uid == uid;
      allow create, update: if request.auth != null && request.auth.uid == uid
        && request.resource.data.size() <= 24
        && request.resource.data.nome is string && request.resource.data.nome.size() <= 24
        && request.resource.data.xp is int && request.resource.data.xp >= 0
        && request.resource.data.nivel is int
        && (!('pts' in request.resource.data) || request.resource.data.pts is int);
    }
    // convites de partida: só quem mandou e quem recebeu podem ver e responder
    match /convites/{id} {
      allow read, delete: if request.auth != null && (resource.data.de == request.auth.uid || resource.data.para == request.auth.uid);
      allow create: if request.auth != null && request.resource.data.de == request.auth.uid
        && request.resource.data.para is string && request.resource.data.para != request.auth.uid;
      allow update: if request.auth != null && (resource.data.de == request.auth.uid || resource.data.para == request.auth.uid)
        && request.resource.data.de == resource.data.de && request.resource.data.para == resource.data.para;
    }
    // pedidos de amizade: o id é "quemPede_quemRecebe"; só quem recebe responde
    match /pedidos/{id} {
      allow read, delete: if request.auth != null && (resource.data.de == request.auth.uid || resource.data.para == request.auth.uid);
      allow create: if request.auth != null && request.resource.data.de == request.auth.uid
        && request.resource.data.para is string && request.resource.data.para != request.auth.uid
        && id == request.auth.uid + '_' + request.resource.data.para;
      allow update: if request.auth != null && resource.data.para == request.auth.uid
        && request.resource.data.de == resource.data.de && request.resource.data.para == resource.data.para;
    }
    // histórico público: as últimas partidas de cada jogador (todos veem, só o dono grava)
    match /historicos/{uid} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == uid;
    }
    // campeões do mês: quem abre o app primeiro no mês novo grava quem era o 1º; ninguém muda depois
    match /campeoes/{mes} {
      allow read: if true;
      allow create: if request.auth != null
        && request.resource.data.keys().hasOnly(['uid', 'nome', 'xp', 't'])
        && request.resource.data.uid is string && request.resource.data.xp is int;
    }
    // guildas: todos veem; os membros atualizam; quem não é membro só pode se incluir numa guilda aberta
    match /guildas/{gid} {
      allow read: if true;
      allow create: if request.auth != null && request.resource.data.lider == request.auth.uid
        && request.resource.data.ids == [request.auth.uid];
      allow update: if request.auth != null && request.resource.data.ids.size() <= 30 && (
        request.auth.uid in resource.data.ids ||
        (resource.data.aberta == true && request.resource.data.lider == resource.data.lider
          && request.resource.data.ids.toSet().difference(resource.data.ids.toSet()) == [request.auth.uid].toSet()
          && resource.data.ids.toSet().difference(request.resource.data.ids.toSet()).size() == 0));
      allow delete: if request.auth != null && resource.data.lider == request.auth.uid;
    }
    // pedidos para entrar numa guilda: o id é "guilda_quemPede"; o líder aceita ou recusa
    match /guildaPedidos/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid
        && id == request.resource.data.g + '_' + request.auth.uid;
      allow delete: if request.auth != null && (resource.data.uid == request.auth.uid
        || get(/databases/$(database)/documents/guildas/$(resource.data.g)).data.lider == request.auth.uid);
    }
    // mural de cada guilda: só os membros leem e escrevem
    match /murais/{gid} {
      allow read, write: if request.auth != null
        && request.auth.uid in get(/databases/$(database)/documents/guildas/$(gid)).data.ids;
    }
    // desafio do dia: cada jogador grava o próprio resultado uma vez só (não dá para mudar depois)
    match /diarios/{data}/r/{uid} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == uid && request.resource.data.pts is int
        && request.resource.data.pts >= 0 && request.resource.data.pts <= 1000;
    }
    // torneio da semana: cada jogador grava e atualiza só o próprio resultado
    match /torneios/{semana}/r/{uid} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null && request.auth.uid == uid && request.resource.data.pts is int
        && request.resource.data.pts >= 0 && request.resource.data.pts <= 5000
        && (resource == null || request.resource.data.pts >= resource.data.pts);
    }
    // salas de partidas online: quem tem conta pode criar e entrar;
    // depois que a sala fecha, só os dois jogadores podem mexer nela
    match /salas/{codigo} {
      allow read, create: if request.auth != null;
      allow update: if request.auth != null && (
        resource.data.status == 'esperando' ||
        request.auth.uid == resource.data.brancasUid ||
        request.auth.uid == resource.data.pretasUid);
    }
  }
}
